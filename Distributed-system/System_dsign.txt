## 1) API Gateway (Express + TypeScript)

**Purpose:** Single entry point; routes to microservices; good place for auth, rate limits, request logging.

**`gateway/src/index.ts`**

```ts
import express from 'express';
import morgan from 'morgan';
import httpProxy from 'http-proxy-middleware';
import helmet from 'helmet';

const app = express();
app.use(helmet());
app.use(express.json());
app.use(morgan('dev'));

// Simple service discovery via env (could be Consul/Eureka/etc.)
const AUTH_URL = process.env.AUTH_URL || 'http://localhost:4001';
const ORDER_URL = process.env.ORDER_URL || 'http://localhost:4002';
const PAYMENT_URL = process.env.PAYMENT_URL || 'http://localhost:4003';

// Pass-through proxies by path prefix
app.use('/auth', httpProxy.createProxyMiddleware({ target: AUTH_URL, changeOrigin: true }));
app.use('/orders', httpProxy.createProxyMiddleware({ target: ORDER_URL, changeOrigin: true }));
app.use('/payments', httpProxy.createProxyMiddleware({ target: PAYMENT_URL, changeOrigin: true }));

app.get('/health', (_req, res) => res.json({ ok: true, ts: Date.now() }));

const port = process.env.PORT || 4000;
app.listen(port, () => console.log(`Gateway on :${port}`));
```

**Install:** `npm i express morgan helmet http-proxy-middleware`

## 3) Load Balancer (Nginx) – Round Robin across replicas

**`infra/nginx/nginx.conf`**

```nginx
worker_processes auto;
events { worker_connections 1024; }
http {
  upstream order_backends {
    server order-svc-1:4002;
    server order-svc-2:4002;
  }

  server {
    listen 80;

    # LB for /orders -> two replicas
    location /orders {
      proxy_pass http://order_backends;
    }

    # Gateway pass-through for others
    location /auth { proxy_pass http://gateway:4000; }
    location /payments { proxy_pass http://gateway:4000; }
    location /health { proxy_pass http://gateway:4000; }
  }
}
```

> Spin multiple container instances (e.g., `order-svc-1`, `order-svc-2`). Nginx will round-robin requests.

## 4) Asynchronous Processing with Queues (Redis + BullMQ)

**Producer (e.g., create job after payment)**

```ts
// services/payment-service/src/queue.ts
import { Queue } from 'bullmq';
const q = new Queue('email', { connection: { host: 'redis', port: 6379 } });
export async function enqueueReceipt(email: string, orderId: string) {
  await q.add('send-receipt', { email, orderId });
}
```

**Worker**

```ts
// workers/email-worker/src/index.ts
import { Worker, QueueScheduler } from 'bullmq';

new QueueScheduler('email', { connection: { host: 'redis', port: 6379 } });

const worker = new Worker('email', async job => {
  if (job.name === 'send-receipt') {
    const { email, orderId } = job.data;
    // send email here
    console.log('📧 sending receipt to', email, 'for order', orderId);
  }
}, { connection: { host: 'redis', port: 6379 } });

worker.on('failed', (job, err) => console.error('DLQ candidate:', job?.id, err));
```

**Usage in payment route**

```ts
// services/payment-service/src/index.ts (fragment)
import { enqueueReceipt } from './queue';
app.post('/', async (req, res) => {
  const { email, orderId } = req.body;
  await enqueueReceipt(email, orderId);
  res.json({ status: 'PAID', enqueued: true });
});
```

## 6) Caching with Redis (per-route cache middleware)

```ts
// gateway/src/cache.ts
import type { Request, Response, NextFunction } from 'express';
import { createClient } from 'redis';

const client = createClient({ url: 'redis://redis:6379' });
client.connect();

export function cache(ttlSec: number) {
  return async (req: Request, res: Response, next: NextFunction) => {
    const key = `cache:${req.method}:${req.originalUrl}`;
    const cached = await client.get(key);
    if (cached) return res.set('X-Cache', 'HIT').type('json').send(cached);

    const send = res.send.bind(res);
    res.send = (body: any) => {
      // store stringified body
      client.setEx(key, ttlSec, typeof body === 'string' ? body : JSON.stringify(body)).catch(()=>{});
      res.set('X-Cache', 'MISS');
      return send(body);
    } as any;
    next();
  };
}
```

**Use it**

```ts
// gateway/src/index.ts (fragment)
import { cache } from './cache';
app.use('/products', cache(30)); // cache product list for 30s
```

## 7) Rate Limiting (Token Bucket in Redis)

```ts
// gateway/src/rateLimit.ts
import type { Request, Response, NextFunction } from 'express';
import { createClient } from 'redis';

const r = createClient({ url: 'redis://redis:6379' });
r.connect();

interface Options { capacity: number; refillPerSec: number; }
export function tokenBucket({ capacity, refillPerSec }: Options) {
  return async (req: Request, res: Response, next: NextFunction) => {
    const id = req.ip; // or API key
    const key = `tb:${id}`;

    const now = Date.now();
    const state = await r.hGetAll(key);

    let tokens = Number(state.tokens ?? capacity);
    let last = Number(state.last ?? now);

    const elapsedSec = (now - last) / 1000;
    tokens = Math.min(capacity, tokens + elapsedSec * refillPerSec);

    if (tokens < 1) {
      const retry = Math.ceil((1 - tokens) / refillPerSec);
      res.set('Retry-After', String(retry));
      return res.status(429).json({ error: 'rate_limited' });
    }

    tokens -= 1;
    await r.hSet(key, { tokens: tokens.toFixed(4), last: String(now) });
    await r.expire(key, 3600);
    next();
  };
}
```

**Use it**

```ts
// gateway/src/index.ts (fragment)
import { tokenBucket } from './rateLimit';
app.use(tokenBucket({ capacity: 20, refillPerSec: 5 }));
```

## 8) Database: Primary + Read Replicas (PostgreSQL example)

```ts
// services/order-service/src/db.ts
import { Pool } from 'pg';

export const writer = new Pool({
  connectionString: process.env.PG_PRIMARY, // e.g. postgres://user:pass@pg-primary:5432/app
});

export const reader = new Pool({
  connectionString: process.env.PG_REPLICA, // e.g. postgres://user:pass@pg-replica:5432/app
});
```

**Usage pattern**

```ts
// write path
await writer.query('INSERT INTO orders(id, user_id, total) VALUES($1,$2,$3)', [id, uid, total]);

// read path
const { rows } = await reader.query('SELECT * FROM orders WHERE user_id = $1 LIMIT 20', [uid]);
```

> Keep writes on the primary; send reads to replicas. For eventual consistency, add `ORDER BY created_at` and tolerate slight replication lag.

## 9) CDN + Caching Headers (Express static)

```ts
// gateway/src/static.ts
import express from 'express';
import path from 'path';

export function mountStatic(app: express.Express) {
  const assets = path.join(process.cwd(), 'public');
  app.use('/static', express.static(assets, {
    setHeaders(res, filePath) {
      if (filePath.endsWith('.jpg') || filePath.endsWith('.png') || filePath.endsWith('.css') || filePath.endsWith('.js')) {
        res.setHeader('Cache-Control', 'public, max-age=31536000, immutable');
      } else {
        res.setHeader('Cache-Control', 'public, max-age=60');
      }
    }
  }));
}
```

> Put a CDN (e.g., CloudFront/Cloudflare) in front of `/static`. The headers help the CDN cache assets near users.

## 11) Circuit Breaker (wrap unstable downstream calls)

```ts
// gateway/src/circuit.ts
import axios from 'axios';

type State = 'CLOSED' | 'OPEN' | 'HALF';
let state: State = 'CLOSED';
let failures = 0;
let openedAt = 0;

const THRESHOLD = 5;       // failures before OPEN
const COOLDOWN_MS = 10000; // try again after 10s

export async function callOrders(url: string) {
  if (state === 'OPEN' && Date.now() - openedAt < COOLDOWN_MS) {
    throw new Error('circuit_open');
  }
  try {
    const res = await axios.get(url, { timeout: 2000 });
    failures = 0; state = 'CLOSED';
    return res.data;
  } catch (e) {
    failures++;
    if (failures >= THRESHOLD) { state = 'OPEN'; openedAt = Date.now(); }
    throw e;
  }
}
```

## 14) Observability (basic)

**Request logging + latency header**

```ts
// gateway/src/obs.ts
import type { Request, Response, NextFunction } from 'express';
export function latency() {
  return (req: Request, res: Response, next: NextFunction) => {
    const t0 = process.hrtime.bigint();
    res.on('finish', () => {
      const ms = Number((process.hrtime.bigint() - t0) / 1000000n);
      res.setHeader('X-Response-Time', `${ms}ms`);
      console.log(`${req.method} ${req.originalUrl} -> ${res.statusCode} ${ms}ms`);
    });
    next();
  };
}
```

**Use it**

```ts
// gateway/src/index.ts (fragment)
import { latency } from './obs';
app.use(latency());
```

## 15) End-to-End Flow (Putting it together)

1. Client hits `POST /payments` at **Gateway**.
2. Gateway forwards to **payment-service**.
3. Payment-service responds quickly and enqueues an email job via **BullMQ** (Queue) and publishes a `payments` event via **Redis Pub/Sub**.
4. **Email worker** consumes the queue and sends the receipt.
5. Other services (e.g., **notifications**) subscribe to the `payments` channel and act independently.
6. Frequently-read endpoints (e.g., `/products`) are cached by the **Redis cache middleware**.
7. **Nginx** load-balances traffic across **order-service** replicas.
8. **HPA** in **Kubernetes** scales replicas based on CPU.
9. **Circuit breaker** in gateway prevents cascading failures when `order-service` is unhealthy.
